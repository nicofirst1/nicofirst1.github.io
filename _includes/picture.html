{% comment %} Responsive picture element with WebP fallback + responsive srcset. External URLs, SVGs, and WebPs are served as plain . -480/-960 WebP variants (generated by scripts/generate_image_variants.sh) are picked up automatically when they exist next to the source image. Pass width/height (intrinsic px) to reserve layout space and avoid CLS. Pass sizes to override the card-sized default (e.g. full-width post heroes). Usage: {% include picture.html src=item.image alt=item.title class="my-class" loading="lazy" fetchpriority="high" width="1000" height="1000" sizes="100vw" %} {% endcomment %} {% capture img_attrs %}{% if include.class %} class="{{ include.class }}"{% endif %}{% if include.loading %} loading="{{ include.loading }}"{% endif %}{% if include.fetchpriority %} fetchpriority="{{ include.fetchpriority }}"{% endif %}{% if include.width %} width="{{ include.width }}"{% endif %}{% if include.height %} height="{{ include.height }}"{% endif %}{% endcapture %} {% assign img_ext = include.src | split: '.' | last %} {% assign is_external = false %} {% if include.src contains 'http://' or include.src contains 'https://' %}{% assign is_external = true %}{% endif %} {% if is_external %} {% elsif img_ext == 'svg' or img_ext == 'webp' %} {% else %} {% assign src_parts = include.src | split: '.' %} {% assign src_base = src_parts | pop | join: '.' %} {% assign webp_src = src_base | append: '.webp' %} {% assign v480_path = src_base | append: '-480.webp' %} {% assign v960_path = src_base | append: '-960.webp' %} {% assign has480 = site.static_files | where: 'path', v480_path | size %} {% assign has960 = site.static_files | where: 'path', v960_path | size %} {% assign sizes = include.sizes | default: '(min-width: 1024px) 420px, (min-width: 640px) 50vw, 100vw' %} {% if has480 > 0 or has960 > 0 %} {% else %} {% endif %} {% endif %}
